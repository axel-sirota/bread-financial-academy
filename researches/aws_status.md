**Cold-start gotcha**: A freshly-created MWAA env takes ~25-40 min to reach AVAILABLE, then ANOTHER ~10-15 min before the scheduler actively parses new DAG files. ### 11.2 Bash one-shot to re-apply IAM grants (idempotent, instructor-01 can run)

```bash
#!/usr/bin/env bash
# Bread Financial Academy - AWS recovery script
# Idempotently re-applies IAM grants required by W13-W22.
# Run as instructor-01 (datacouch profile). Re-runs are safe.
set -euo pipefail
export AWS_PROFILE=datacouch
export AWS_DEFAULT_REGION=us-west-2

ACCOUNT_ID=962804699607
SAGEMAKER_ROLE_ARN=arn:aws:iam::${ACCOUNT_ID}:role/SageMakerStudentExecutionRole
SNS_ALERTS_ARN=arn:aws:sns:us-west-2:${ACCOUNT_ID}:bread-academy-class-alerts

echo "==> 1. Confirm cohort-2 and cohort-3 do NOT have AWSDenyAll attached"
for g in cohort-2 cohort-3; do
  attached=$(aws iam list-attached-group-policies \
    --group-name "breadfinancial-labsupport-academy-${g}" \
    --query 'AttachedPolicies[].PolicyName' --output text)
  if echo "$attached" | grep -q AWSDenyAll; then
    echo "  DETACHING AWSDenyAll from $g"
    aws iam detach-group-policy \
      --group-name "breadfinancial-labsupport-academy-${g}" \
      --policy-arn arn:aws:iam::aws:policy/AWSDenyAll
  else
    echo "  $g: AWSDenyAll already absent"
  fi
done

echo "==> 2. Ensure MWAAExecutionRole has both inline policies"
mwaa_inline=$(aws iam list-role-policies --role-name MWAAExecutionRole \
  --query 'PolicyNames' --output text)
for p in MWAAExecutionPolicy MWAAWeek21Week22ActionsPolicy; do
  if echo "$mwaa_inline" | grep -qw "$p"; then
    echo "  $p: present"
  else
    echo "  $p: MISSING - apply from /tmp/${p}.json"
    # Document the policy JSON elsewhere; this script only checks.
  fi
done

echo "==> 3. Verify SageMakerStudentExecutionRole attached policies"
expected_attached="BreadFinancialCourseExtras AmazonEC2FullAccess CloudWatchLogsFullAccess AmazonSageMakerFullAccess AmazonS3FullAccess"
sm_attached=$(aws iam list-attached-role-policies --role-name SageMakerStudentExecutionRole \
  --query 'AttachedPolicies[].PolicyName' --output text)
for p in $expected_attached; do
  if echo "$sm_attached" | grep -qw "$p"; then
    echo "  $p: attached"
  else
    echo "  $p: MISSING - attach with iam:AttachRolePolicy"
  fi
done

echo "==> 4. Detach BarclaysS3Access inline if present (S3FullAccess covers it)"
if aws iam get-role-policy --role-name SageMakerStudentExecutionRole \
   --policy-name BarclaysS3Access >/dev/null 2>&1; then
  echo "  DETACHING BarclaysS3Access"
  aws iam delete-role-policy --role-name SageMakerStudentExecutionRole \
    --policy-name BarclaysS3Access
else
  echo "  BarclaysS3Access already absent"
fi

echo "==> 5. Confirm all 60 students are in BreadFinancialStudents"
for n in $(seq -w 1 60); do
  groups=$(aws iam list-groups-for-user --user-name "student-${n}" \
    --query 'Groups[].GroupName' --output text 2>/dev/null || echo "")
  if echo "$groups" | grep -qw BreadFinancialStudents; then
    : # ok
  else
    echo "  student-${n}: NOT in BreadFinancialStudents - adding"
    aws iam add-user-to-group --user-name "student-${n}" \
      --group-name BreadFinancialStudents 2>&1 | head -1
  fi
done

echo "==> 6. SNS topic exists + subscription"
aws sns get-topic-attributes --topic-arn "$SNS_ALERTS_ARN" >/dev/null
echo "  $SNS_ALERTS_ARN: present"

echo "==> 7. Pretrained model artifact exists"
aws s3 head-object --bucket bread-academy-shared --key pretrained/model.tar.gz >/dev/null
echo "  s3://bread-academy-shared/pretrained/model.tar.gz: present"

echo "==> 8. MWAA env AVAILABLE"
status=$(aws mwaa get-environment --name bread-academy-airflow \
  --query 'Environment.Status' --output text)
echo "  bread-academy-airflow: $status"

echo "==> 9. SageMaker endpoint InService"
status=$(aws sagemaker describe-endpoint --endpoint-name fraud-classifier-endpoint \
  --query 'EndpointStatus' --output text)
echo "  fraud-classifier-endpoint: $status"

echo "==> 10. Seed CloudWatch FraudClassifier/Accuracy with a synthetic point"
aws cloudwatch put-metric-data --namespace FraudClassifier \

echo "  FraudClassifier/Accuracy: 0.92 published"
```
